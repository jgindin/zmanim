לְדָוִד ׀ יְהֹוָה ׀ אוֹרִי וְיִשְׁעִי, מִמִּי אִירָא, יְהֹוָה מָעוֹז חַיַּי, מִמִּי אֶפְחָֽד: בִּקְרֹב עָלַי ׀ מְרֵעִים לֶֽאֱכֹל אֶת־בְּשָׂרִי, צָרַי וְאֹֽיְבַי לִי, הֵֽמָּה כָֽשְׁלוּ וְנָפָֽלוּ: אִם־תַּֽחֲנֶה עָלַי ׀ מַֽחֲנֶה לֹֽא־יִירָא לִבִּי, אִם־תָּקוּם עָלַי מִלְחָמָה בְּזֹאת אֲנִי בוֹטֵֽחַ: אַחַת ׀ שָׁאַֽלְתִּי מֵֽאֵת־יְהֹוָה אוֹתָהּ אֲבַקֵּשׁ, שִׁבְתִּֽי בְּבֵית־יְהֹוָה כׇּל־יְמֵי חַיַּי, לַֽחֲזוֹת בְּנֹֽעַם־יְהֹוָה וּלְבַקֵּר בְּהֵיכָלֽוֹ: כִּי יִצְפְּנֵֽנִי ׀ בְּסֻכֹּה בְּיוֹם רָעָה, יַסְתִּרֵֽנִי בְּסֵֽתֶר אָֽהֳלוֹ, בְּצוּר יְרֽוֹמְמֵֽנִי: וְעַתָּה יָרוּם רֹאשִׁי עַל־אֹֽיְבַי סְבִֽיבוֹתַי, וְאֶזְבְּחָה בְאָֽהֳלוֹ זִבְחֵי תְרוּעָה, אָשִֽׁירָה וַֽאֲזַמְּרָה לַיהֹוָֽה: שְׁמַע־יְהֹוָה קוֹלִי אֶקְרָא וְחׇנֵּֽנִי וַֽעֲנֵֽנִי: לְךָ ׀ אָמַר לִבִּי בַּקְּשׁוּ פָנָי, אֶת־פָּנֶֽיךָ יְהֹוָה אֲבַקֵּֽשׁ: אַל־תַּסְתֵּר פָּנֶֽיךָ ׀ מִמֶּֽנִּי, אַל־תַּט בְּאַף עַבְדֶּֽךָ, עֶזְרָתִֽי הָיִֽיתָ, אַל־תִּטְּשֵֽׁנִי וְאַל־תַּֽעַזְבֵֽנִי אֱלֹהֵי יִשְׁעִֽי: כִּי־אָבִי וְאִמִּי עֲזָבֽוּנִי, וַֽיהֹוָה יַֽאַסְפֵֽנִי: הוֹרֵֽנִי יְהֹוָה דַּרְכֶּֽךָ, וּנְחֵֽנִי בְּאֹֽרַח מִישׁוֹר לְמַֽעַן שֽׁוֹרְרָֽי: אַֽל־תִּתְּנֵֽנִי בְּנֶֽפֶשׁ צָרָי, כִּי קָֽמוּ־בִי עֵֽדֵי־שֶֽׁקֶר וִיפֵֽחַ חָמָֽס: לוּלֵא הֶֽאֱמַֽנְתִּי לִרְאוֹת בְּטֽוּב־יְהֹוָה בְּאֶֽרֶץ חַיִּֽים: קַוֵּה אֶל־יְהֹוָה, חֲזַק וְיַֽאֲמֵץ לִבֶּֽךָ וְקַוֵּה אֶל־יְהֹוָה:
